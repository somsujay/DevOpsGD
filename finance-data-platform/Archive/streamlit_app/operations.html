Scripts Reference

Deployment

# Deploy all migrations to an environment
bash scripts/deploy_schemachange.sh --env=dev
bash scripts/deploy_schemachange.sh --env=qa
bash scripts/deploy_schemachange.sh --env=prod

# Dry-run (shows what would be deployed)
bash scripts/deploy_schemachange.sh --env=prod --dry-run

Data Loading

# Historical load (full reset + initial data)
bash scripts/run_historical.sh --env=dev                  # CSV source (default)
bash scripts/run_historical.sh --env=dev --source=iceberg # Iceberg/Parquet source

# Incremental load (requires historical load first)
bash scripts/run_incremental.sh --env=dev
bash scripts/run_incremental.sh --env=dev --source=iceberg

Testing

# Smoke tests (schema/object existence checks)
bash scripts/run_smoke_tests.sh --env=dev

# Integration tests executed via CI after smoke tests pass

Teardown & Rollback

# Preview what will be dropped
bash scripts/drop_objects.sh

# Actually drop (DESTRUCTIVE)
bash scripts/drop_objects.sh --confirm

# Rollback to a previous version
bash scripts/rollback.sh --env=preprod --version=v1.2.0
bash scripts/rollback.sh --env=prod --version=abc123f

Dashboard

bash scripts/streamlit_start.sh    # Start at http://localhost:8501
bash scripts/streamlit_stop.sh     # Stop

Adding a New Migration

# 1. Create file in appropriate subdirectory
banking/clean/retail/V2.050.200__add_customer_segment.sql

# 2. Use Jinja variables for portability
USE DATABASE {{ database }};
ALTER TABLE {{ clean_schema }}.DIMCUSTOMER ADD COLUMN SEGMENT VARCHAR(50);

# 3. Test with dry-run
bash scripts/deploy_schemachange.sh --env=dev --dry-run

# 4. Apply
bash scripts/deploy_schemachange.sh --env=dev

Data Flow

Source Files (CSV/Parquet) | v RAW (Raw Landing) | T_CUSTOMER, T_ACCOUNT, T_TRANSACTION | Loaded via: Stage -> Stream -> Tasks (CSV) or COPY INTO (Parquet) | v CLEAN (Conformed Dimensions) | DIMCUSTOMER (SCD-2), DIMACCOUNT (SCD-1) | DIMTRANSACTIONTYPE, DIMDATE | v CONFORMED (Business Facts) FACTDAILYTRANSACTION, FACTDAILYAGG Views: MonthlySpendProfile, TxnTypeTrend

Adding a New Environment

# 1. Add to environments.yml:
staging:
  database: FINANCE_CORE_DEV
  schema_suffix: _STAGING
  raw_schema: RAW_STAGING
  clean_schema: CLEAN_STAGING
  conformed_schema: CONFORMED_STAGING
  governance_schema: GOVERNANCE_STAGING
  warehouse: COMPUTE_WH
  connection: MY_TRIAL_ACCOUNT

# 2. Deploy (creates schemas and applies all migrations):
bash scripts/deploy_schemachange.sh --env=staging

Key-Pair Authentication Setup

# 1. Generate RSA key pair
openssl genrsa 2048 | openssl pkcs8 -topk8 -inform PEM -out ~/.snowflake/trial_key.p8 -nocrypt
openssl rsa -in ~/.snowflake/trial_key.p8 -pubout -out ~/.snowflake/trial_key.pub
chmod 600 ~/.snowflake/trial_key.p8

# 2. Register public key in Snowflake
ALTER USER SOMSUJAY SET RSA_PUBLIC_KEY='<paste public key without headers>';

# 3. Configure ~/.snowflake/connections.toml:
[MY_TRIAL_ACCOUNT]
account = "KXAXARZ-GW22129"
user = "SOMSUJAY"
authenticator = "SNOWFLAKE_JWT"
private_key_path = "/path/to/.snowflake/trial_key.p8"
warehouse = "COMPUTE_WH"
database = "FINANCE_CORE_DEV"
